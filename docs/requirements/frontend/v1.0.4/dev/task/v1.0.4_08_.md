## 实现要点

### 1. 页面组件

```tsx
// pages/Chat/settings/index.tsx
import React, { useState, useEffect } from 'react';
import { Button, Switch, Radio, Card, message } from 'antd';
import { ArrowLeftOutlined } from '@ant-design/icons';
import { history } from '@umijs/max';
import styles from './index.less';

const SettingsPage: React.FC = () => {
  // 从 localStorage 读取设置
  const [darkMode, setDarkMode] = useState(() => {
    return localStorage.getItem('darkMode') === 'true';
  });
  const [enterToSend, setEnterToSend] = useState(() => {
    return localStorage.getItem('enterToSend') !== 'false';
  });

  // 保存深色模式设置
  const handleDarkModeChange = (checked: boolean) => {
    setDarkMode(checked);
    localStorage.setItem('darkMode', String(checked));
    message.success(checked ? '已开启深色模式' : '已关闭深色模式');
    // TODO: 实际切换主题
  };

  // 保存 Enter 发送设置
  const handleEnterToSendChange = (e: any) => {
    const value = e.target.value === 'enter';
    setEnterToSend(value);
    localStorage.setItem('enterToSend', String(value));
    message.success(value ? '已设置为 Enter 发送' : '已设置为 Ctrl+Enter 发送');
  };

  return (
    <div className={styles.container}>
      {/* 头部 */}
      <header className={styles.header}>
        <div className={styles.headerLeft}>
          <Button 
            type="text" 
            icon={<ArrowLeftOutlined />} 
            onClick={() => history.push('/chat')}
          >
            返回
          </Button>
          <h1 className={styles.title}>设置</h1>
        </div>
      </header>

      {/* 内容区 */}
      <div className={styles.content}>
        {/* 应用设置 */}
        <Card className={styles.card}>
          <h2 className={styles.sectionTitle}>应用设置</h2>
          
          <div className={styles.settingItem}>
            <div className={styles.settingLabel}>
              <span>深色模式</span>
              <span className={styles.settingDescription}>开启后界面将切换为深色主题</span>
            </div>
            <Switch 
              checked={darkMode} 
              onChange={handleDarkModeChange}
            />
          </div>

          <div className={styles.settingDivider} />

          <div className={styles.settingItem}>
            <div className={styles.settingLabel}>
              <span>发送消息</span>
              <span className={styles.settingDescription}>选择发送消息的快捷键</span>
            </div>
            <Radio.Group
              value={enterToSend ? 'enter' : 'ctrlEnter'}
              onChange={handleEnterToSendChange}
            >
              <Radio value="enter">Enter</Radio>
              <Radio value="ctrlEnter">Ctrl+Enter</Radio>
            </Radio.Group>
          </div>
        </Card>

        {/* 关于 */}
        <Card className={styles.card}>
          <h2 className={styles.sectionTitle}>关于</h2>
          
          <div className={styles.aboutItem}>
            <span className={styles.aboutLabel}>版本</span>
            <span className={styles.aboutValue}>v1.0.4</span>
          </div>

          <div className={styles.aboutItem}>
            <span className={styles.aboutLabel}>应用名称</span>
            <span className={styles.aboutValue}>OPC-CAT</span>
          </div>

          <div className={styles.aboutItem}>
            <span className={styles.aboutLabel}>描述</span>
            <span className={styles.aboutValue}>AI 智能助手</span>
          </div>
        </Card>
      </div>
    </div>
  );
};

export default SettingsPage;
```
