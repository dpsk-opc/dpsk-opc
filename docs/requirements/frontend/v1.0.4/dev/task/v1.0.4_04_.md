## 实现要点

### 1. Agent 列表页主组件

```tsx
// pages/Chat/agents/index.tsx
import React, { useState, useEffect, useMemo } from 'react';
import { Input, Button, Select, Spin, message } from 'antd';
import { PlusOutlined, ArrowLeftOutlined } from '@ant-design/icons';
import { history } from '@umijs/max';
import AgentCard from '@/components/AgentCard';
import { AgentService, handleAgentError } from '@/services/agentService';
import type { Agent, Team } from '@/types/agent';
import styles from './index.less';

const AgentListPage: React.FC = () => {
  const [agents, setAgents] = useState<Agent[]>([]);
  const [teams, setTeams] = useState<Team[]>([]);
  const [loading, setLoading] = useState(true);
  const [searchKeyword, setSearchKeyword] = useState('');
  const [selectedTeam, setSelectedTeam] = useState<string | undefined>();

  // 获取 Agent 列表和 Team 列表
  const fetchData = async () => {
    setLoading(true);
    try {
      const [agentsData, teamsData] = await Promise.all([
        AgentService.list(selectedTeam),
        TeamService.list(),
      ]);
      setAgents(agentsData);
      setTeams(teamsData);
    } catch (error) {
      message.error(handleAgentError(error));
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    fetchData();
  }, [selectedTeam]);

  // 按 Team 分组
  const groupedAgents = useMemo(() => {
    const filtered = searchKeyword
      ? agents.filter(a => a.name?.includes(searchKeyword) || a.agent_id.includes(searchKeyword))
      : agents;
    
    const grouped: { team?: string; agents: Agent[] }[] = [];
    const teamsMap = new Map<string, Agent[]>();
    const noTeam: Agent[] = [];

    filtered.forEach(agent => {
      if (agent.team) {
        const list = teamsMap.get(agent.team) || [];
        list.push(agent);
        teamsMap.set(agent.team, list);
      } else {
        noTeam.push(agent);
      }
    });

    if (noTeam.length > 0) {
      grouped.push({ agents: noTeam });
    }
    
    teamsMap.forEach((list, team) => {
      grouped.push({ team, agents: list });
    });

    return grouped;
  }, [agents, searchKeyword]);

  return (
    <div className={styles.container}>
      {/* 头部 */}
      <header className={styles.header}>
        <div className={styles.headerLeft}>
          <Button 
            type="text" 
            icon={<ArrowLeftOutlined />} 
            onClick={() => history.push('/chat')}
          >
            返回
          </Button>
          <h1 className={styles.title}>Agent 管理</h1>
        </div>
        <Button 
          type="primary" 
          icon={<PlusOutlined />}
          onClick={() => history.push('/chat/agents/new')}
        >
          新建
        </Button>
      </header>

      {/* 搜索和筛选 */}
      <div className={styles.toolbar}>
        <Input.Search
          placeholder="搜索 Agent..."
          value={searchKeyword}
          onChange={(e) => setSearchKeyword(e.target.value)}
          className={styles.searchInput}
        />
        <Select
          placeholder="全部"
          value={selectedTeam}
          onChange={setSelectedTeam}
          allowClear
          className={styles.teamSelect}
          options={[
            { label: `全部 (${agents.length})`, value: undefined },
            ...teams.map(t => ({ label: `${t.name} (${t.agent_count})`, value: t.name })),
          ]}
        />
      </div>

      {/* 内容区 */}
      <div className={styles.content}>
        {loading ? (
          <div className={styles.loading}>
            <Spin size="large" />
          </div>
        ) : groupedAgents.length === 0 ? (
          <EmptyState
            icon="🤖"
            title="还没有 Agent"
            description="点击「新建」开始使用"
            action={{
              text: '新建 Agent',
              onClick: () => history.push('/chat/agents/new'),
            }}
          />
        ) : (
          <div className={styles.list}>
            {groupedAgents.map((group) => (
              <div key={group.team || 'no-team'} className={styles.group}>
                {group.team && (
                  <div className={styles.groupTitle}>
                    <span className={styles.groupIcon}>👥</span>
                    {group.team}
                  </div>
                )}
                {group.agents.map((agent) => (
                  <AgentCard
                    key={agent.agent_id}
                    agent={agent}
                    onClick={() => history.push(`/chat/agents/${encodeURIComponent(agent.agent_id)}`)}
                  />
                ))}
              </div>
            ))}
          </div>
        )}
      </div>
    </div>
  );
};

export default AgentListPage;
```

### 2. AgentCard 组件

```tsx
// components/AgentCard/index.tsx
import React from 'react';
import { RightOutlined } from '@ant-design/icons';
import type { Agent } from '@/types/agent';
import styles from './index.less';

interface AgentCardProps {
  agent: Agent;
  onClick?: () => void;
}

const AgentCard: React.FC<AgentCardProps> = ({ agent, onClick }) => {
  return (
    <div className={styles.card} onClick={onClick}>
      <div className={styles.icon}>🤖</div>
      <div className={styles.info}>
        <div className={styles.name}>{agent.name || agent.agent_id}</div>
        <div className={styles.team}>
          {agent.team || '没有分类'}
        </div>
      </div>
      <RightOutlined className={styles.arrow} />
    </div>
  );
};

export default AgentCard;
```
