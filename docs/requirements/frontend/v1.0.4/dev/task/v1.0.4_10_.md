### AgentService 测试

```typescript
// services/agentService.test.ts
import { AgentService, TeamService } from './agentService';

// Mock fetch
global.fetch = jest.fn();

describe('AgentService', () => {
  beforeEach(() => {
    jest.clearAllMocks();
  });

  describe('list', () => {
    it('should return agents list', async () => {
      const mockAgents = [
        { agent_id: 'agent1', name: 'Agent 1' },
        { agent_id: 'agent2', name: 'Agent 2' },
      ];
      (global.fetch as jest.Mock).mockResolvedValueOnce({
        ok: true,
        json: () => Promise.resolve(mockAgents),
      });

      const result = await AgentService.list();
      expect(result).toEqual(mockAgents);
      expect(fetch).toHaveBeenCalledWith(
        expect.stringContaining('/agents/')
      );
    });

    it('should filter by team', async () => {
      const mockAgents = [{ agent_id: 'agent1', team: 'engineering' }];
      (global.fetch as jest.Mock).mockResolvedValueOnce({
        ok: true,
        json: () => Promise.resolve(mockAgents),
      });

      await AgentService.list('engineering');
      expect(fetch).toHaveBeenCalledWith(
        expect.stringContaining('team=engineering')
      );
    });

    it('should throw error on failure', async () => {
      (global.fetch as jest.Mock).mockResolvedValueOnce({
        ok: false,
        status: 500,
      });

      await expect(AgentService.list()).rejects.toThrow('获取 Agent 列表失败: 500');
    });
  });

  describe('get', () => {
    it('should return agent detail', async () => {
      const mockDetail = {
        agent_id: 'agent1',
        markdown: '# Agent 1\n\nContent here',
      };
      (global.fetch as jest.Mock).mockResolvedValueOnce({
        ok: true,
        json: () => Promise.resolve(mockDetail),
      });

      const result = await AgentService.get('agent1');
      expect(result).toEqual(mockDetail);
    });

    it('should throw 404 error', async () => {
      (global.fetch as jest.Mock).mockResolvedValueOnce({
        ok: false,
        status: 404,
      });

      await expect(AgentService.get('nonexistent')).rejects.toThrow('Agent 不存在');
    });
  });

  describe('create', () => {
    it('should create agent successfully', async () => {
      const newAgent = { agent_id: 'new-agent', markdown: '# New' };
      (global.fetch as jest.Mock).mockResolvedValueOnce({
        ok: true,
        json: () => Promise.resolve(newAgent),
      });

      const result = await AgentService.create(newAgent);
      expect(result).toEqual(newAgent);
      expect(fetch).toHaveBeenCalledWith(
        expect.stringContaining('/agents/'),
        expect.objectContaining({ method: 'POST' })
      );
    });

    it('should throw 409 on duplicate', async () => {
      (global.fetch as jest.Mock).mockResolvedValueOnce({
        ok: false,
        status: 409,
      });

      await expect(
        AgentService.create({ agent_id: 'existing', markdown: '#' })
      ).rejects.toThrow('该 Agent ID 已存在');
    });
  });

  describe('update', () => {
    it('should update agent', async () => {
      (global.fetch as jest.Mock).mockResolvedValueOnce({
        ok: true,
        json: () => Promise.resolve({ agent_id: 'agent1' }),
      });

      await AgentService.update('agent1', '# Updated');
      expect(fetch).toHaveBeenCalledWith(
        expect.stringContaining('agent1'),
        expect.objectContaining({ method: 'PUT' })
      );
    });
  });

  describe('delete', () => {
    it('should delete agent', async () => {
      (global.fetch as jest.Mock).mockResolvedValueOnce({
        ok: true,
      });

      await AgentService.delete('agent1');
      expect(fetch).toHaveBeenCalledWith(
        expect.stringContaining('agent1'),
        expect.objectContaining({ method: 'DELETE' })
      );
    });
  });
});

describe('TeamService', () => {
  beforeEach(() => {
    jest.clearAllMocks();
  });

  describe('list', () => {
    it('should return teams', async () => {
      const mockTeams = [
        { name: 'engineering', path: 'engineering', agent_count: 5 },
      ];
      (global.fetch as jest.Mock).mockResolvedValueOnce({
        ok: true,
        json: () => Promise.resolve(mockTeams),
      });

      const result = await TeamService.list();
      expect(result).toEqual(mockTeams);
    });
  });

  describe('create', () => {
    it('should create team', async () => {
      const mockTeam = { name: 'new-team', path: 'new-team', agent_count: 0 };
      (global.fetch as jest.Mock).mockResolvedValueOnce({
        ok: true,
        json: () => Promise.resolve(mockTeam),
      });

      const result = await TeamService.create('new-team');
      expect(result).toEqual(mockTeam);
    });
  });

  describe('delete', () => {
    it('should delete team', async () => {
      (global.fetch as jest.Mock).mockResolvedValueOnce({
        ok: true,
      });

      await TeamService.delete('old-team');
      expect(fetch).toHaveBeenCalledWith(
        expect.stringContaining('old-team'),
        expect.objectContaining({ method: 'DELETE' })
      );
    });

    it('should delete with force', async () => {
      (global.fetch as jest.Mock).mockResolvedValueOnce({
        ok: true,
      });

      await TeamService.delete('team-with-agents', true);
      expect(fetch).toHaveBeenCalledWith(
        expect.stringContaining('force=true'),
        expect.objectContaining({ method: 'DELETE' })
      );
    });
  });
});
```

### BasicLayout 测试

```tsx
// layouts/BasicLayout/BasicLayout.test.tsx
import React from 'react';
import { render, screen, fireEvent, waitFor } from '@testing-library/react';
import { BasicLayout, SidebarContext } from './index';

const mockHistory = {
  push: jest.fn(),
  location: { pathname: '/chat' },
};

jest.mock('@umijs/max', () => ({
  history: mockHistory,
}));

describe('BasicLayout', () => {
  it('should render children', () => {
    render(
      <BasicLayout>
        <div data-testid="child">Child Content</div>
      </BasicLayout>
    );
    expect(screen.getByTestId('child')).toBeInTheDocument();
  });

  it('should render hamburger menu', () => {
    render(<BasicLayout><div /></BasicLayout>);
    expect(screen.getByRole('button')).toBeInTheDocument();
  });

  it('should open sidebar when hamburger clicked', () => {
    render(<BasicLayout><div /></BasicLayout>);
    const hamburger = screen.getByRole('button');
    fireEvent.click(hamburger);
    expect(screen.getByText('Agent 管理')).toBeInTheDocument();
  });
});

describe('SidebarContext', () => {
  it('should provide default values', () => {
    let contextValue = { open: false, setOpen: () => {} };
    render(
      <SidebarContext.Consumer>
        {(value) => {
          contextValue = value;
          return null;
        }}
      </SidebarContext.Consumer>
    );
    expect(contextValue.open).toBe(false);
    expect(typeof contextValue.setOpen).toBe('function');
  });
});
```
