### 1. Markdown 编辑器封装

使用 `@uiw/react-md-editor` 库封装编辑器组件：

```tsx
// components/MarkdownEditor/index.tsx
import React, { useState } from 'react';
import MDEditor from '@uiw/react-md-editor';
import '@uiw/react-md-editor/markdown-editor.css';
import '@uiw/react-codemirror/lib/editor/index.css';

interface MarkdownEditorProps {
  value?: string;
  onChange?: (value: string) => void;
  minHeight?: number;
  placeholder?: string;
}

const MarkdownEditor: React.FC<MarkdownEditorProps> = ({
  value = '',
  onChange,
  minHeight = 300,
  placeholder,
}) => {
  return (
    <div data-color-mode="light">
      <MDEditor
        value={value}
        onChange={onChange}
        height={minHeight}
        preview="edit"
        textareaProps={{
          placeholder: placeholder || '输入 Markdown 内容...',
        }}
      />
    </div>
  );
};

export default MarkdownEditor;
```

### 2. Team 选择器组件

```tsx
// components/TeamSelector/index.tsx
import React, { useState, useEffect } from 'react';
import { Select, Button, Input, Modal, message } from 'antd';
import { PlusOutlined } from '@ant-design/icons';
import { TeamService } from '@/services/agentService';
import type { Team } from '@/types/agent';

interface TeamSelectorProps {
  value?: string;
  onChange?: (value: string | undefined) => void;
}

const TeamSelector: React.FC<TeamSelectorProps> = ({ value, onChange }) => {
  const [teams, setTeams] = useState<Team[]>([]);
  const [loading, setLoading] = useState(false);
  const [modalOpen, setModalOpen] = useState(false);
  const [newTeamName, setNewTeamName] = useState('');
  const [creating, setCreating] = useState(false);

  // 获取 Team 列表
  const fetchTeams = async () => {
    setLoading(true);
    try {
      const data = await TeamService.list();
      setTeams(data);
    } catch {
      // 忽略错误，使用空列表
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    fetchTeams();
  }, []);

  // 创建新 Team
  const handleCreate = async () => {
    if (!newTeamName.trim()) {
      message.error('请输入 Team 名称');
      return;
    }
    setCreating(true);
    try {
      const newTeam = await TeamService.create(newTeamName.trim());
      setTeams([...teams, newTeam]);
      onChange?.(newTeam.name);
      setNewTeamName('');
      setModalOpen(false);
      message.success('创建成功');
    } catch {
      message.error('创建失败');
    } finally {
      setCreating(false);
    }
  };

  return (
    <>
      <Select
        value={value}
        onChange={onChange}
        placeholder="选择 Team（可选）"
        allowClear
        loading={loading}
        options={teams.map(t => ({
          label: `${t.name} (${t.agent_count})`,
          value: t.name,
        }))]
        dropdownRender={(menu) => (
          <>
            {menu}
            <div className={styles.dropdownFooter}>
              <Button 
                type="link" 
                icon={<PlusOutlined />} 
                onClick={() => setModalOpen(true)}
                size="small"
              >
                新建 Team
              </Button>
            </div>
          </>
        )}
      />

      <Modal
        title="新建 Team"
        open={modalOpen}
        onCancel={() => {
          setModalOpen(false);
          setNewTeamName('');
        }}
        onOk={handleCreate}
        confirmLoading={creating}
      >
        <Input
          value={newTeamName}
          onChange={(e) => setNewTeamName(e.target.value)}
          placeholder="输入 Team 名称"
          onPressEnter={handleCreate}
        />
        <p className={styles.hint}>支持嵌套，如 engineering/backend</p>
      </Modal>
    </>
  );
};

export default TeamSelector;
```

### 3. Agent 表单组件

```tsx
// components/AgentForm/index.tsx
import React, { useState, useEffect } from 'react';
import { Form, Input, Button, Spin, message } from 'antd';
import { ArrowLeftOutlined, SaveOutlined } from '@ant-design/icons';
import { history, useParams } from '@umijs/max';
import MarkdownEditor from '@/components/MarkdownEditor';
import TeamSelector from '@/components/TeamSelector';
import { AgentService, handleAgentError } from '@/services/agentService';
import type { AgentDetail } from '@/types/agent';
import styles from './form.less';

interface AgentFormProps {
  mode: 'create' | 'edit';
}

const AgentForm: React.FC<AgentFormProps> = ({ mode }) => {
  const { id } = useParams<{ id: string }>();
  const [form] = Form.useForm();
  const [markdown, setMarkdown] = useState('');
  const [loading, setLoading] = useState(mode === 'edit');
  const [submitting, setSubmitting] = useState(false);
  const [agent, setAgent] = useState<AgentDetail | null>(null);

  const isEdit = mode === 'edit';

  // 加载已有数据
  useEffect(() => {
    if (isEdit && id) {
      const fetchAgent = async () => {
        setLoading(true);
        try {
          const data = await AgentService.get(decodeURIComponent(id));
          setAgent(data);
          setMarkdown(data.markdown);
          // 提取 frontmatter 设置表单初始值
          const frontmatter = extractFrontmatter(data.markdown);
          form.setFieldsValue({
            agent_id: data.agent_id,
            team: frontmatter.team,
            name: frontmatter.name,
            description: frontmatter.description,
          });
        } catch (error) {
          message.error(handleAgentError(error));
        } finally {
          setLoading(false);
        }
      };
      fetchAgent();
    }
  }, [isEdit, id]);

  // 提交表单
  const handleSubmit = async () => {
    try {
      await form.validateFields();
    } catch {
      return;
    }

    const values = form.getFieldsValue();
    const finalMarkdown = buildMarkdown(values, markdown);

    setSubmitting(true);
    try {
      if (isEdit && id) {
        await AgentService.update(decodeURIComponent(id), finalMarkdown);
        message.success('更新成功');
      } else {
        await AgentService.create({
          agent_id: values.agent_id,
          markdown: finalMarkdown,
        });
        message.success('创建成功');
      }
      history.push('/chat/agents');
    } catch (error) {
      message.error(handleAgentError(error));
    } finally {
      setSubmitting(false);
    }
  };

  // 提取 frontmatter
  const extractFrontmatter = (md: string) => {
    const match = md.match(/^---\n([\s\S]*?)\n---/);
    if (!match) return {};
    const result: Record<string, string> = {};
    match[1].split('\n').forEach(line => {
      const [key, ...valueParts] = line.split(':');
      if (key && valueParts.length) {
        result[key.trim()] = valueParts.join(':').trim();
      }
    });
    return result;
  };

  // 构建包含 frontmatter 的 Markdown
  const buildMarkdown = (values: Record<string, string>, content: string) => {
    const frontmatter = [
      '---',
      `name: ${values.name || ''}`,
      `description: ${values.description || ''}`,
      `team: ${values.team || ''}`,
      '---',
    ].join('\n');
    
    // 保留原有内容（去掉旧的 frontmatter）
    const contentWithoutOld = content.replace(/^---\n[\s\S]*?\n---\n*/, '');
    
    return `${frontmatter}\n\n${contentWithoutOld}`;
  };

  if (loading) {
    return (
      <div className={styles.container}>
        <div className={styles.loading}>
          <Spin size="large" />
        </div>
      </div>
    );
  }

  return (
    <div className={styles.container}>
      {/* 头部 */}
      <header className={styles.header}>
        <div className={styles.headerLeft}>
          <Button 
            type="text" 
            icon={<ArrowLeftOutlined />} 
            onClick={() => history.push(isEdit ? `/chat/agents/${id}` : '/chat/agents')}
          >
            返回
          </Button>
          <h1 className={styles.title}>{isEdit ? '编辑 Agent' : '新建 Agent'}</h1>
        </div>
        <Button 
          type="primary"
          icon={<SaveOutlined />}
          loading={submitting}
          onClick={handleSubmit}
        >
          保存
        </Button>
      </header>

      {/* 表单 */}
      <div className={styles.form}>
        <Form
          form={form}
          layout="vertical"
          initialValues={{
            name: '',
            description: '',
            team: undefined,
            agent_id: '',
          }}
        >
          <Form.Item
            name="agent_id"
            label="Agent ID"
            rules={[
              { required: true, message: '请输入 Agent ID' },
              { pattern: /^[\w\/]+$/, message: '仅支持字母、数字、下划线和路径分隔符' },
            ]}
            extra="唯一标识，创建后不可修改"
          >
            <Input 
              placeholder="如 engineering/frontend"
              disabled={isEdit}
            />
          </Form.Item>

          <Form.Item
            name="team"
            label="Team"
            extra="Agent 所属部门"
          >
            <TeamSelector />
          </Form.Item>

          <Form.Item
            name="name"
            label="名称"
            extra="Agent 显示名称"
          >
            <Input placeholder="如 前端工程师" />
          </Form.Item>

          <Form.Item
            name="description"
            label="描述"
            extra="Agent 功能描述"
          >
            <Input.TextArea placeholder="简要描述 Agent 的功能..." rows={2} />
          </Form.Item>

          <Form.Item
            label="Markdown 内容"
            required
            extra="Agent 的系统提示词，支持 Markdown 语法"
          >
            <MarkdownEditor
              value={markdown}
              onChange={setMarkdown}
              minHeight={400}
            />
          </Form.Item>
        </Form>
      </div>
    </div>
  );
};

export default AgentForm;
```
