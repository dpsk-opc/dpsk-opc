## 实现要点

### 1. 详情页主组件

```tsx
// pages/Chat/agents/$id.tsx
import React, { useState, useEffect } from 'react';
import { Button, Spin, message, Modal } from 'antd';
import { ArrowLeftOutlined, EditOutlined, DeleteOutlined, CopyOutlined } from '@ant-design/icons';
import { history, useParams } from '@umijs/max';
import { AgentService, handleAgentError } from '@/services/agentService';
import type { AgentDetail } from '@/types/agent';
import styles from './$id.less';

const AgentDetailPage: React.FC = () => {
  const { id } = useParams<{ id: string }>();
  const [agent, setAgent] = useState<AgentDetail | null>(null);
  const [loading, setLoading] = useState(true);
  const [deleteModalOpen, setDeleteModalOpen] = useState(false);
  const [deleting, setDeleting] = useState(false);

  // 获取 Agent 详情
  const fetchAgent = async () => {
    if (!id) return;
    setLoading(true);
    try {
      const data = await AgentService.get(decodeURIComponent(id));
      setAgent(data);
    } catch (error) {
      message.error(handleAgentError(error));
      // 延迟返回列表
      setTimeout(() => history.push('/chat/agents'), 2000);
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    fetchAgent();
  }, [id]);

  // 复制 Markdown 内容
  const handleCopy = async () => {
    if (!agent?.markdown) return;
    try {
      await navigator.clipboard.writeText(agent.markdown);
      message.success('已复制到剪贴板');
    } catch {
      message.error('复制失败');
    }
  };

  // 删除 Agent
  const handleDelete = async () => {
    if (!id) return;
    setDeleting(true);
    try {
      await AgentService.delete(decodeURIComponent(id));
      message.success('删除成功');
      history.push('/chat/agents');
    } catch (error) {
      message.error(handleAgentError(error));
    } finally {
      setDeleting(false);
      setDeleteModalOpen(false);
    }
  };

  // 从 frontmatter 提取信息
  const extractFrontmatter = (markdown: string) => {
    const match = markdown.match(/^---\n([\s\S]*?)\n---/);
    if (!match) return {};
    const result: Record<string, string> = {};
    match[1].split('\n').forEach(line => {
      const [key, ...valueParts] = line.split(':');
      if (key && valueParts.length) {
        result[key.trim()] = valueParts.join(':').trim();
      }
    });
    return result;
  };

  if (loading) {
    return (
      <div className={styles.container}>
        <div className={styles.loading}>
          <Spin size="large" />
        </div>
      </div>
    );
  }

  if (!agent) {
    return null;
  }

  const frontmatter = extractFrontmatter(agent.markdown);

  return (
    <div className={styles.container}>
      {/* 头部 */}
      <header className={styles.header}>
        <div className={styles.headerLeft}>
          <Button 
            type="text" 
            icon={<ArrowLeftOutlined />} 
            onClick={() => history.push('/chat/agents')}
          >
            返回
          </Button>
          <h1 className={styles.title}>{frontmatter.name || agent.agent_id}</h1>
        </div>
        <Button 
          type="primary"
          icon={<EditOutlined />}
          onClick={() => history.push(`/chat/agents/${id}/edit`)}
        >
          编辑
        </Button>
      </header>

      {/* 内容区 */}
      <div className={styles.content}>
        <div className={styles.infoCard}>
          <div className={styles.infoRow}>
            <span className={styles.infoLabel}>Agent ID:</span>
            <span className={styles.infoValue}>{agent.agent_id}</span>
          </div>
          <div className={styles.infoRow}>
            <span className={styles.infoLabel}>Team:</span>
            <span className={styles.infoValue}>{frontmatter.team || '-'}</span>
          </div>
          
          <div className={styles.divider} />
          
          <div className={styles.markdownSection}>
            <div className={styles.markdownHeader}>
              <span className={styles.markdownTitle}>Markdown 内容预览</span>
              <Button 
                type="text" 
                icon={<CopyOutlined />}
                onClick={handleCopy}
                size="small"
              >
                复制
              </Button>
            </div>
            <pre className={styles.markdownContent}>
              <code>{agent.markdown}</code>
            </pre>
          </div>
        </div>

        {/* 删除按钮 */}
        <div className={styles.actions}>
          <Button 
            danger
            icon={<DeleteOutlined />}
            onClick={() => setDeleteModalOpen(true)}
          >
            删除 Agent
          </Button>
        </div>
      </div>

      {/* 删除确认弹窗 */}
      <Modal
        title="确认删除"
        open={deleteModalOpen}
        onCancel={() => setDeleteModalOpen(false)}
        footer={[
          <Button key="cancel" onClick={() => setDeleteModalOpen(false)}>
            取消
          </Button>,
          <Button 
            key="delete" 
            type="primary" 
            danger 
            loading={deleting}
            onClick={handleDelete}
          >
            确认删除
          </Button>,
        ]}
      >
        <p>确定要删除「{frontmatter.name || agent.agent_id}」吗？</p>
        <p>此操作无法撤销。</p>
      </Modal>
    </div>
  );
};

export default AgentDetailPage;
```
