### 2. API 服务 (services/agentService.ts)

```typescript
import type {
  Agent,
  AgentDetail,
  Team,
  CreateAgentRequest,
  UpdateAgentRequest,
  CreateTeamRequest,
} from '@/types/agent';

// API Base URL
const API_BASE_URL = 'http://localhost:8000/plugins/admin/api';

/**
 * Agent 服务
 */
export const AgentService = {
  /**
   * 获取所有 Agent
   */
  list: async (team?: string): Promise<Agent[]> => {
    const url = team 
      ? `${API_BASE_URL}/agents/?team=${encodeURIComponent(team)}`
      : `${API_BASE_URL}/agents/`;
    const response = await fetch(url);
    if (!response.ok) throw new Error(`获取 Agent 列表失败: ${response.status}`);
    return response.json();
  },

  /**
   * 获取单个 Agent 详情
   */
  get: async (agentId: string): Promise<AgentDetail> => {
    const response = await fetch(
      `${API_BASE_URL}/agents/${encodeURIComponent(agentId)}`
    );
    if (!response.ok) {
      if (response.status === 404) throw new Error('Agent 不存在');
      throw new Error(`获取 Agent 详情失败: ${response.status}`);
    }
    return response.json();
  },

  /**
   * 创建 Agent
   */
  create: async (data: CreateAgentRequest): Promise<Agent> => {
    const response = await fetch(`${API_BASE_URL}/agents/`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(data),
    });
    if (!response.ok) {
      if (response.status === 409) throw new Error('该 Agent ID 已存在');
      throw new Error(`创建 Agent 失败: ${response.status}`);
    }
    return response.json();
  },

  /**
   * 更新 Agent
   */
  update: async (agentId: string, markdown: string): Promise<Agent> => {
    const response = await fetch(
      `${API_BASE_URL}/agents/${encodeURIComponent(agentId)}`,
      {
        method: 'PUT',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ markdown }),
      }
    );
    if (!response.ok) {
      throw new Error(`更新 Agent 失败: ${response.status}`);
    }
    return response.json();
  },

  /**
   * 删除 Agent
   */
  delete: async (agentId: string): Promise<void> => {
    const response = await fetch(
      `${API_BASE_URL}/agents/${encodeURIComponent(agentId)}`,
      { method: 'DELETE' }
    );
    if (!response.ok) {
      throw new Error(`删除 Agent 失败: ${response.status}`);
    }
  },
};

/**
 * Team 服务
 */
export const TeamService = {
  /**
   * 获取所有 Team
   */
  list: async (): Promise<Team[]> => {
    const response = await fetch(`${API_BASE_URL}/agents/teams/`);
    if (!response.ok) throw new Error(`获取 Team 列表失败: ${response.status}`);
    return response.json();
  },

  /**
   * 创建 Team
   */
  create: async (teamName: string): Promise<Team> => {
    const response = await fetch(`${API_BASE_URL}/agents/teams/`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ team_name: teamName }),
    });
    if (!response.ok) throw new Error(`创建 Team 失败: ${response.status}`);
    return response.json();
  },

  /**
   * 删除 Team
   */
  delete: async (teamName: string, force?: boolean): Promise<void> => {
    const url = force
      ? `${API_BASE_URL}/agents/teams/${teamName}?force=true`
      : `${API_BASE_URL}/agents/teams/${teamName}`;
    const response = await fetch(url, { method: 'DELETE' });
    if (!response.ok) {
      throw new Error(`删除 Team 失败: ${response.status}`);
    }
  },
};

/**
 * 错误处理工具
 */
export const handleAgentError = (error: unknown): string => {
  if (error instanceof Error) {
    if (error.message.includes('404')) return 'Agent 不存在或已被删除';
    if (error.message.includes('409')) return '该 Agent ID 已存在';
    return error.message;
  }
  return '操作失败，请稍后重试';
};
```
