## 实现要点

### 1. 页面组件

```tsx
// pages/Chat/history/index.tsx
import React, { useState, useEffect } from 'react';
import { Button, Spin, message, List } from 'antd';
import { ArrowLeftOutlined, MessageOutlined } from '@ant-design/icons';
import { history } from '@umijs/max';
import { ChatService } from '@/services/chatService';
import type { SessionHistory } from '@/services/chatService';
import styles from './index.less';

const HistoryPage: React.FC = () => {
  const [sessions, setSessions] = useState<SessionHistory[]>([]);
  const [loading, setLoading] = useState(true);

  // 格式化时间
  const formatTime = (timestamp: number): string => {
    const date = new Date(timestamp);
    const now = new Date();
    const today = new Date(now.getFullYear(), now.getMonth(), now.getDate());
    const yesterday = new Date(today.getTime() - 24 * 60 * 60 * 1000);
    const sessionDate = new Date(date.getFullYear(), date.getMonth(), date.getDate());

    if (sessionDate.getTime() === today.getTime()) {
      return `今天 ${date.toLocaleTimeString('zh-CN', { hour: '2-digit', minute: '2-digit' })}`;
    } else if (sessionDate.getTime() === yesterday.getTime()) {
      return `昨天 ${date.toLocaleTimeString('zh-CN', { hour: '2-digit', minute: '2-digit' })}`;
    } else {
      return date.toLocaleDateString('zh-CN', { month: 'long', day: 'numeric' }) + 
             ' ' + 
             date.toLocaleTimeString('zh-CN', { hour: '2-digit', minute: '2-digit' });
    }
  };

  // 获取会话列表
  const fetchSessions = async () => {
    setLoading(true);
    try {
      // TODO: 实际调用获取会话列表 API
      // 目前使用模拟数据
      setSessions([]);
    } catch (error) {
      message.error('获取会话历史失败');
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    fetchSessions();
  }, []);

  return (
    <div className={styles.container}>
      {/* 头部 */}
      <header className={styles.header}>
        <div className={styles.headerLeft}>
          <Button 
            type="text" 
            icon={<ArrowLeftOutlined />} 
            onClick={() => history.push('/chat')}
          >
            返回
          </Button>
          <h1 className={styles.title}>会话历史</h1>
        </div>
      </header>

      {/* 内容区 */}
      <div className={styles.content}>
        {loading ? (
          <div className={styles.loading}>
            <Spin size="large" />
          </div>
        ) : sessions.length === 0 ? (
          <div className={styles.empty}>
            <MessageOutlined className={styles.emptyIcon} />
            <p className={styles.emptyText}>暂无历史会话</p>
          </div>
        ) : (
          <List
            className={styles.list}
            dataSource={sessions}
            renderItem={(session) => (
              <List.Item 
                className={styles.item}
                onClick={() => history.push(`/chat?session=${session.session_id}`)}
              >
                <div className={styles.itemIcon}>🤖</div>
                <div className={styles.itemContent}>
                  <div className={styles.itemPreview}>
                    {session.messages[0]?.content?.slice(0, 50) || '空会话'}...
                  </div>
                  <div className={styles.itemTime}>
                    {formatTime(session.created_at)}
                  </div>
                </div>
              </List.Item>
            )}
          />
        )}
      </div>
    </div>
  );
};

export default HistoryPage;
```
